# What the server actually needs to run. `requirements.txt` includes this file
# and adds the development and evaluation extras on top, so nothing is pinned in
# two places and the two cannot drift.
#
# The split exists because the deployed image was installing the whole evaluation
# harness and the test runner for no reason. scipy alone is around 60 MB and is
# imported by exactly two files, `backend/evaluation/penn_action.py` and
# `scripts/eval_penn_action.py`, neither of which the API touches - verified by
# importing `backend.main` and checking sys.modules.
#
# Python 3.13 (see README for why not 3.11).

# Core pipeline (Phases A-B)
mediapipe==0.10.35
numpy==2.2.6

# Pinned even though mediapipe pulls it in anyway: an unpinned transitive
# dependency in the container would make the image non-reproducible for no gain.
matplotlib==3.10.9

# Annotated-video encoding. Ships a static ffmpeg binary per
# platform, so there is no "install ffmpeg first" step. OpenCV on its own can
# only write MPEG-4 Part 2 here, which no browser will play.
imageio-ffmpeg==0.6.0

# API server
fastapi==0.136.3
uvicorn[standard]==0.49.0
python-multipart==0.0.32

# NOTE: OpenCV is deliberately absent. mediapipe requires opencv-contrib-python
# and pip installs it regardless of what is written here. Development keeps that
# GUI build (pinned in requirements.txt); the container swaps it for
# opencv-contrib-python-headless at the same version, which needs no system
# libraries. See the Dockerfile for the evidence that the swap is safe.
